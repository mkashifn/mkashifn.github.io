Under the hood

SERP-grounded outline generation

Research the live SERP, then draft an outline the model must ground in real competitor headings and entity gaps — not its imagination.

engine/outline.py PYTHON 
from anthropic import Anthropic
from serp import fetch_top_results, extract_entities

client = Anthropic()
MODEL  = "claude-sonnet-4-20250514"

def build_outline(keyword: str, brand_voice: str) -> dict:
    # 1. Ground the model in the live SERP, not guesses
    top      = fetch_top_results(keyword, n=10)
    headings = [h for r in top for h in r.headings]
    entities = extract_entities(top)          # entities competitors cover

    prompt = f"""You are an SEO content strategist.
Keyword: {keyword}
Competitor headings: {headings}
Entities to cover: {entities}
Write a JSON outline (H2/H3) that covers every entity above,
fills gaps competitors miss, and matches this brand voice: {brand_voice}.
Reply with ONLY JSON."""

    msg = client.messages.create(
        model=MODEL, max_tokens=1500, temperature=0.3,
        messages=[{"role": "user", "content": prompt}],
    )
    return parse_outline(msg.content[0].text)   # validated downstream
